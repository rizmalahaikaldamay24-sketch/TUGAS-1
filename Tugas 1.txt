#include <iostream>
#include <string>  // Library untuk menggunakan tipe data string (nama & NIM)
#include <iomanip> // Library untuk memformat 2 angka di belakang koma

using namespace std;

int main() {
    // Deklarasi variabel identitas
    string nama, nim;
    
    // Deklarasi variabel nilai
    double tugas, kuis, uts, uas, nilai_akhir;
    string grade, status;

    cout << "=========================================" << endl;
    cout << "    PROGRAM HITUNG NILAI AKHIR MAHASISWA " << endl;
    cout << "=========================================" << endl;

    // Input Identitas Mahasiswa
    cout << "Masukkan Nama Mahasiswa : ";
    getline(cin, nama); // Menggunakan getline agar bisa membaca nama dengan spasi
    
    cout << "Masukkan NIM Mahasiswa  : ";
    cin >> nim;

    // Input Nilai Komponen
    cout << "Masukkan Nilai Tugas    : ";
    cin >> tugas;
    cout << "Masukkan Nilai Kuis     : ";
    cin >> kuis;
    cout << "Masukkan Nilai UTS      : ";
    cin >> uts;
    cout << "Masukkan Nilai UAS      : ";
    cin >> uas;

    // Menghitung Nilai Akhir (Tugas 20%, Kuis 10%, UTS 30%, UAS 40%)
    nilai_akhir = (tugas * 0.20) + (kuis * 0.10) + (uts * 0.30) + (uas * 0.40);

    // Menentukan Grade dan Status
    if (nilai_akhir >= 85) {
        grade = "A";
        status = "LULUS";
    } else if (nilai_akhir >= 70) {
        grade = "B";
        status = "LULUS";
    } else if (nilai_akhir >= 55) {
        grade = "C";
        status = "LULUS";
    } else if (nilai_akhir >= 40) {
        grade = "D";
        status = "TIDAK LULUS";
    } else {
        grade = "E";
        status = "TIDAK LULUS";
    }

    // Menampilkan Hasil Rapor / Transkrip Nilai
    cout << "\n=========================================" << endl;
    cout << "         HASIL EVALUASI AKADEMIK         " << endl;
    cout << "=========================================" << endl;
    cout << "Nama Mahasiswa : " << nama << endl;
    cout << "NIM            : " << nim << endl;
    cout << "-----------------------------------------" << endl;
    cout << "Nilai Akhir    : " << fixed << setprecision(2) << nilai_akhir << endl;
    cout << "Grade          : " << grade << endl;
    cout << "Status         : " << status << endl;
    cout << "=========================================" << endl;

    return 0;
}
